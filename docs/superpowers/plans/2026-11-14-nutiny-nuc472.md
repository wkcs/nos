#include <assert.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>

/* 模拟 ARMv7-M SCB->VTOR 与 MPU 寄存器阵列的测试桩 */
static uint32_t g_test_vtor;
static uint32_t g_test_rnr;
static uint32_t g_test_rbar[8];
static uint32_t g_test_rasr[8];
static uint32_t g_test_mpu_ctrl;

#define ARMV7M_MPU_TEST_HOST 1
#define ARMV7M_TEST_SCB_VTOR g_test_vtor
#define ARMV7M_TEST_MPU_CTRL g_test_mpu_ctrl
#define ARMV7M_TEST_MPU_RNR  g_test_rnr
#define ARMV7M_TEST_MPU_RBAR (g_test_rbar[g_test_rnr & 0x7U])
#define ARMV7M_TEST_MPU_RASR (g_test_rasr[g_test_rnr & 0x7U])

#include "../../arch/arm/mm/mpu.c"

static void reset_mpu_regs(uint32_t vtor)
{
    g_test_vtor = vtor;
    g_test_rnr = 0U;
    g_test_mpu_ctrl = 0U;
    memset(g_test_rbar, 0, sizeof(g_test_rbar));
    memset(g_test_rasr, 0, sizeof(g_test_rasr));
}

int main(void)
{
#ifdef TEST_NUC472_LAYOUT
    reset_mpu_regs(0x00000000UL);
    arch_mm_init_static_regions_for_test();
    /* NUC472 的 VTOR=0 且 Flash 位于 0x00000000，禁止启用 Region 0 空指针陷阱 */
    assert((g_test_rasr[NOS_MPU_REGION_NULL_TRAP] & 1U) == 0U);
    /* Region 1: 0x00000000, 512 KiB (log2(512K)-1 = 18), 可读可执行 */
    assert((g_test_rbar[NOS_MPU_REGION_FLASH] & 0xFFFFFFE0U) == 0x00000000UL);
    assert((g_test_rasr[NOS_MPU_REGION_FLASH] & 1U) == 1U);
    assert(((g_test_rasr[NOS_MPU_REGION_FLASH] >> 1) & 0x1FU) == 18U);
    assert(((g_test_rasr[NOS_MPU_REGION_FLASH] >> 28) & 1U) == 0U);
    /* Region 2: 0x20000000, 64 KiB (log2(64K)-1 = 15), XN=1 */
    assert((g_test_rbar[NOS_MPU_REGION_SRAM] & 0xFFFFFFE0U) == 0x20000000UL);
    assert((g_test_rasr[NOS_MPU_REGION_SRAM] & 1U) == 1U);
    assert(((g_test_rasr[NOS_MPU_REGION_SRAM] >> 1) & 0x1FU) == 15U);
    assert(((g_test_rasr[NOS_MPU_REGION_SRAM] >> 28) & 1U) == 1U);
    printf("NUC472 MPU static layout test passed\n");
#else
    reset_mpu_regs(0x08000000UL);
    arch_mm_init_static_regions_for_test();
    /* STM32 默认 VTOR != 0 且 Flash != 0，Region 0 空指针陷阱生效 */
    assert((g_test_rasr[NOS_MPU_REGION_NULL_TRAP] & 1U) == 1U);
    assert((g_test_rbar[NOS_MPU_REGION_FLASH] & 0xFFFFFFE0U) == 0x08000000UL);
    assert(((g_test_rasr[NOS_MPU_REGION_FLASH] >> 1) & 0x1FU) == 19U);
    assert((g_test_rbar[NOS_MPU_REGION_SRAM] & 0xFFFFFFE0U) == 0x20000000UL);
    assert(((g_test_rasr[NOS_MPU_REGION_SRAM] >> 1) & 0x1FU) == 16U);
    printf("STM32 MPU static layout test passed\n");
#endif
    return 0;
}

{
    /* 仅当 VTOR 与 Flash 均不在 0x00000000 时设置 Region 0 空指针陷阱，

/**
 * Copyright (C) 2026 Nick Hu <huqihan@live.com>
 *
 * NUC472 时钟初始化与系统频率计算 (12 MHz HXT -> 84 MHz PLL，失败自动回退 HIRC)
 */
#include <soc/nuc472_soc.h>

#ifndef NUC472_HOST_TEST
uint32_t SystemCoreClock = 22118400UL;
#else
static uint32_t g_test_core_clk = 22118400UL;
#define SystemCoreClock g_test_core_clk
#endif

static uint32_t g_hclk_hz = 22118400UL;
static uint32_t g_uart0_clk_hz = 22118400UL;

static bool clk_wait_ready(uint32_t status_mask)
{
    uint32_t timeout = 400000UL;
    while (timeout-- > 0UL) {
        if ((NUC472_CLK->STATUS & status_mask) == status_mask) {
            return true;
        }
    }
    return false;
}

uint32_t nuc472_clk_init_84mhz(void)
{
    uint32_t sys_hclk = NUC472_HIRC_HZ;

    nuc472_sys_unlock();

    /* 保持片内 HIRC 运行作为安全兜底时钟 */
    NUC472_CLK->PWRCTL |= NUC472_CLK_PWRCTL_HIRCEN;
    (void)clk_wait_ready(NUC472_CLK_STATUS_HIRCSTB);
    NUC472_CLK->CLKSEL0 = (NUC472_CLK->CLKSEL0 & ~NUC472_CLK_CLKSEL0_HCLKSEL_Msk) |
                          NUC472_CLK_CLKSEL0_HCLKSEL_HIRC;

    /* 启动外部 12 MHz HXT 与片内 10 kHz LIRC(用于 WDT)；LXT 仅在需要 RTC 精确日历时按键启用 */
    NUC472_CLK->PWRCTL |= NUC472_CLK_PWRCTL_HXTEN | NUC472_CLK_PWRCTL_LIRCEN;
    if (clk_wait_ready(NUC472_CLK_STATUS_HXTSTB)) {
        /* 84 MHz HCLK 需先关闭或配置 Flash 存取周期安全模式 (FTCTL.FOM = 000 为 >72 MHz) */
        NUC472_FMC->FTCTL &= ~NUC472_FMC_FTCTL_FOM_Msk;

        /* 12 MHz / 2 * 28 / 2 = 84 MHz (NUC472_CLK_PLLCTL_84MHZ_HXT = 0x0000401AUL) */
        NUC472_CLK->PLLCTL = NUC472_CLK_PLLCTL_84MHZ_HXT;
        if (clk_wait_ready(NUC472_CLK_STATUS_PLLSTB)) {
            NUC472_CLK->CLKDIV0 = (NUC472_CLK->CLKDIV0 &
                                   ~(NUC472_CLK_CLKDIV0_HCLKDIV_Msk | NUC472_CLK_CLKDIV0_UARTDIV_Msk));
            NUC472_CLK->CLKSEL0 = (NUC472_CLK->CLKSEL0 & ~NUC472_CLK_CLKSEL0_HCLKSEL_Msk) |
                                  NUC472_CLK_CLKSEL0_HCLKSEL_PLL;
            /* UART0 选择 12 MHz HXT 时钟源，115200 bps 误差低且不受 PLL 分频影响 */
            NUC472_CLK->CLKSEL1 = (NUC472_CLK->CLKSEL1 & ~NUC472_CLK_CLKSEL1_UARTSEL_Msk) |
                                  NUC472_CLK_CLKSEL1_UARTSEL_HXT;
            sys_hclk = NUC472_PLL_84MHZ_HZ;
            g_uart0_clk_hz = NUC472_HXT_HZ;
        }
    }

    if (sys_hclk != NUC472_PLL_84MHZ_HZ) {
        /* 回退 HIRC (22.1184 MHz)，UART0 同步切到 HIRC 避免波特率失真 */
        NUC472_CLK->CLKSEL0 = (NUC472_CLK->CLKSEL0 & ~NUC472_CLK_CLKSEL0_HCLKSEL_Msk) |
                              NUC472_CLK_CLKSEL0_HCLKSEL_HIRC;
        NUC472_CLK->CLKSEL1 = (NUC472_CLK->CLKSEL1 & ~NUC472_CLK_CLKSEL1_UARTSEL_Msk) |
                              NUC472_CLK_CLKSEL1_UARTSEL_HIRC;
        g_uart0_clk_hz = NUC472_HIRC_HZ;
    }

    nuc472_sys_lock();

    g_hclk_hz = sys_hclk;
    SystemCoreClock = sys_hclk;
    return sys_hclk;
}

uint32_t nuc472_clk_get_hclk_hz(void)
{
    return g_hclk_hz;
}

uint32_t nuc472_clk_get_pclk_hz(void)
{
    /* CLKSEL0.PCLKSEL 默认为 HCLK (1:1) */
    return (NUC472_CLK->CLKSEL0 & NUC472_CLK_CLKSEL0_PCLKSEL_Msk) ? (g_hclk_hz / 2UL) : g_hclk_hz;
}

uint32_t nuc472_clk_get_uart0_hz(void)
{
    return g_uart0_clk_hz;
}

 * Copyright (C) 2026 Nick Hu <huqihan@live.com>
 *
 * NUC472 片内 FMC ISP 底层原语 (含 SRAM .ramfunc 执行与 MPU 安全切换)

#include <kernel/irq.h>
#include <kernel/errno.h>

#define ARMV7M_MPU_CTRL_REG (*(volatile uint32_t *)0xE000ED94UL)

#ifndef NUC472_HOST_TEST
__attribute__((section(".ramfunc"), noinline))
static int fmc_isp_trigger_ramfunc(void)
{
    uint32_t timeout = 1000000UL;
    NUC472_FMC->ISPTRG = NUC472_FMC_ISPTRG_ISPGO;
    /* 双重内存/指令屏障确保 CPU 从 SRAM 等待 ISP 状态清零 */
    __asm__ volatile ("isb\n dsb\n" ::: "memory");
    while ((NUC472_FMC->ISPTRG & NUC472_FMC_ISPTRG_ISPGO) != 0UL) {
        if (--timeout == 0UL) {
            return -ETIMEDOUT;
        }
    }
    return (NUC472_FMC->ISPCTL & NUC472_FMC_ISPCTL_ISPFF) ? -EIO : 0;
}
#else
static int fmc_isp_trigger_ramfunc(void); /* 由测试桩拦截 */
#endif

static int fmc_exec_isp(uint32_t cmd, uint32_t addr, uint32_t dat_in,
                        uint32_t *dat_out, bool aprom_update)
{
    addr_t irq_flags;
    uint32_t mpu_ctrl_saved = 0U;
    int ret;

    irq_flags = disable_irq_save();
#ifndef NUC472_HOST_TEST
    /* 因 SRAM 默认设置 XN=1，执行 .ramfunc 前在关中断下临时关闭 MPU 并恢复 */
    mpu_ctrl_saved = ARMV7M_MPU_CTRL_REG;
    if ((mpu_ctrl_saved & 1UL) != 0UL) {
        ARMV7M_MPU_CTRL_REG = mpu_ctrl_saved & ~1UL;
        __asm__ volatile ("dsb\n isb\n" ::: "memory");
    }
#endif
    nuc472_sys_unlock();
    NUC472_CLK->AHBCLK |= NUC472_CLK_AHBCLK_ISPCKEN;
    NUC472_FMC->ISPCTL |= NUC472_FMC_ISPCTL_ISPEN | NUC472_FMC_ISPCTL_ISPFF;
    if (aprom_update) {
        NUC472_FMC->ISPCTL |= NUC472_FMC_ISPCTL_APUEN;
    } else {
        NUC472_FMC->ISPCTL &= ~NUC472_FMC_ISPCTL_APUEN;
    }
    NUC472_FMC->ISPCMD = cmd;
    NUC472_FMC->ISPADDR = addr;
    NUC472_FMC->ISPDAT = dat_in;

    ret = fmc_isp_trigger_ramfunc();
    if (ret == 0 && dat_out != NULL) {
        *dat_out = NUC472_FMC->ISPDAT;
    }

    NUC472_FMC->ISPCTL &= ~NUC472_FMC_ISPCTL_APUEN;
    nuc472_sys_lock();
#ifndef NUC472_HOST_TEST
    if ((mpu_ctrl_saved & 1UL) != 0UL) {
        ARMV7M_MPU_CTRL_REG = mpu_ctrl_saved;
        __asm__ volatile ("dsb\n isb\n" ::: "memory");
    }
#else
    (void)mpu_ctrl_saved;
#endif
    enable_irq_save(irq_flags);
    return ret;
}

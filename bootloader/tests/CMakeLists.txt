cmake_minimum_required(VERSION 3.20)
# Low-address Flash emulation requires an x86_64 process on macOS (Rosetta on Apple Silicon).
if(APPLE AND NOT DEFINED CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES x86_64 CACHE STRING "Host regression architecture")
endif()
project(bootloader_host_tests C CXX)
enable_testing()
add_executable(test_nlk_bl test_nlk_bl.c fake_hardware.c ../nlk/nlk_bl.c)
target_include_directories(test_nlk_bl PRIVATE stubs ../include ../nlk)
target_compile_definitions(test_nlk_bl PRIVATE BL_VER_MAJOR=1 BL_VER_MINOR=0 BL_VER_PATCH=0)
target_compile_options(test_nlk_bl PRIVATE -Wall -Wextra -Werror)
add_test(NAME nlk_bootloader COMMAND test_nlk_bl)
if(APPLE)
    target_link_options(test_nlk_bl PRIVATE -Wl,-pagezero_size,0x4000)
endif()
add_executable(test_usb test_usb.c ../usb/app/src/usbd_winusb.c)
target_include_directories(test_usb PRIVATE stubs ../include ../nlk ../drivers)
target_compile_options(test_usb PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
add_test(NAME bootloader_usb COMMAND test_usb)
if(NLK_CLI_SOURCE_DIR)
    add_executable(test_cli test_cli.cpp fake_hardware.c ../nlk/nlk_bl.c
        ${NLK_CLI_SOURCE_DIR}/src/client.cpp
        ${NLK_CLI_SOURCE_DIR}/src/caps/system.cpp
        ${NLK_CLI_SOURCE_DIR}/src/caps/flash.cpp
        ${NLK_CLI_SOURCE_DIR}/src/transport_libusb.cpp
        ${NLK_CLI_SOURCE_DIR}/src/transport_serial.cpp)
    target_include_directories(test_cli PRIVATE stubs ../include ../nlk ${NLK_CLI_SOURCE_DIR}/include)
    target_compile_definitions(test_cli PRIVATE BL_VER_MAJOR=1 BL_VER_MINOR=0 BL_VER_PATCH=0)
    target_compile_features(test_cli PRIVATE cxx_std_20)
    target_link_libraries(test_cli PRIVATE ${CMAKE_DL_LIBS})
    if(APPLE)
        target_link_options(test_cli PRIVATE -Wl,-pagezero_size,0x4000)
    endif()
    add_test(NAME cli_bootloader_interop COMMAND test_cli)
endif()
add_executable(test_usb_desc test_usb_desc.c ../usb/app/src/usbd_desc.c)
target_include_directories(test_usb_desc PRIVATE stubs ../usb/app/inc)
target_compile_options(test_usb_desc PRIVATE -Wall -Wextra -Werror)
add_test(NAME bootloader_usb_identity COMMAND test_usb_desc)

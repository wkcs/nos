cmake_minimum_required(VERSION 3.16)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../components")
if(IDF_TARGET STREQUAL "esp32s3" AND EXISTS "$ENV{NOS_ESP_USB_PATH}/device/esp_tinyusb/CMakeLists.txt")
    list(APPEND EXTRA_COMPONENT_DIRS "$ENV{NOS_ESP_USB_PATH}/device/esp_tinyusb")
endif()
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
option(NOS_IDF_ECOSYSTEM "Retain broad SDK entry points for a link audit" OFF)
option(NOS_IDF_REGRESSION "Build the hardware-mutating regression application" OFF)
option(NOS_IDF_STORAGE_WL "Use the distinct wear-levelled storage format" OFF)
idf_build_set_property(NOS_IDF_STORAGE_WL "${NOS_IDF_STORAGE_WL}")
idf_build_set_property(NOS_IDF_REGRESSION "${NOS_IDF_REGRESSION}")
idf_build_set_property(NOS_IDF_ECOSYSTEM "${NOS_IDF_ECOSYSTEM}")
set(NOS_IDF_FILESYSTEM "native2" CACHE STRING "Native snapshot format: native2 or snapshot3")
if(NOT NOS_IDF_FILESYSTEM MATCHES "^(native2|snapshot3)$")
    message(FATAL_ERROR "Unknown NOS_IDF_FILESYSTEM")
endif()
if(NOS_IDF_FILESYSTEM STREQUAL "snapshot3")
    add_compile_definitions(NOS_VFS_SNAPSHOT3=1)
endif()
idf_build_set_property(NOS_ESP_VFS ON)
project(nos_${IDF_TARGET})

include("${CMAKE_CURRENT_LIST_DIR}/../cmake/hardware.cmake")

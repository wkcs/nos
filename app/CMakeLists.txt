set(APP_SRCS "")
set(APP_PRIVATE_INCLUDES "")

include("${CMAKE_CURRENT_SOURCE_DIR}/server/sources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/shell/sources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/lua/sources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/lvgl/sources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/tests/sources.cmake")

if(APP_SRCS)
    add_library(nos_app OBJECT ${APP_SRCS})
    if(APP_PRIVATE_INCLUDES)
        target_include_directories(nos_app PRIVATE ${APP_PRIVATE_INCLUDES})
    endif()
    nos_configure_kernel_target(nos_app)
    if(NOS_APP_OPTIMIZATION_FLAG)
        include(CheckCCompilerFlag)
        check_c_compiler_flag("${NOS_APP_OPTIMIZATION_FLAG}" NOS_HAS_APP_OPTIMIZATION_FLAG)
        if(NOS_HAS_APP_OPTIMIZATION_FLAG)
            target_compile_options(nos_app PRIVATE "${NOS_APP_OPTIMIZATION_FLAG}")
        endif()
    endif()
    if(NOS_APP_COMPILE_OPTIONS)
        target_compile_options(nos_app PRIVATE ${NOS_APP_COMPILE_OPTIONS})
    endif()
else()
    add_library(nos_app INTERFACE)
endif()
